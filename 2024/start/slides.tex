  \begin{frame}{Forward and Backward Stepwise Selection}
    \textbf{Forward} Stepwise Selection:
    \begin{itemize}
      \item Start with \textbf{none} of the predictors
      \item At each step, add the single best predictor
      \item Stop when no additional predictor would be significant
    \end{itemize}
    \vfill
    \textbf{Backward} Stepwise Selection:
    \begin{itemize}
      \item Start with \textbf{all} of the predictors
      \item At each step, drop the least significant predictor
      \item Stop when all remaining predictors are significant
    \end{itemize}
  \end{frame}

  \begin{frame}{Best Subset Selection}
    $$Y \approx \beta_0 + \beta_1 X_1 + \beta_2 X_2 + ... + \beta_p X_p$$
    \vfill
    \begin{align*}
      \hat\beta =& \arg\min_\beta \left\{ \sum_{i = 1}^n \left(y_i - \beta_0 - \sum_{j = 1}^p \beta_j x_{ij}\right)^2 \right\}\\
      & \text{subject to } \sum_{j = 1}^p \mathbb{I}\{\beta_j \ne 0\} \le s
    \end{align*}
    \vfill
    \begin{itemize}
      \item Computationally expensive
      \item What happens to the error of the model as $s$ increases?
    \end{itemize}
  \end{frame}

  \begin{frame}{Convex Optimization}
    \begin{align*}
      \hat\beta =& \arg\min_\beta \left\{ \sum_{i = 1}^n \left(y_i - \beta_0 - \sum_{j = 1}^p \beta_j x_{ij}\right)^2 \right\}\\
      & \text{subject to } \sum_{j = 1}^p {\color{blue} \mathbb{I}\{\beta_j \ne 0\}} \le s \quad \longleftarrow \text{hard \frownie{}}
    \end{align*}
    \vfill
    \begin{align*}
      \hat\beta =& \arg\min_\beta \left\{ \sum_{i = 1}^n \left(y_i - \beta_0 - \sum_{j = 1}^p \beta_j x_{ij}\right)^2 \right\}\\
      & \text{subject to } \sum_{j = 1}^p {\color{blue} |\beta_j|} \le s \quad \longleftarrow \text{easy \smiley{}}
    \end{align*}
  \end{frame}

  \begin{frame}{The Lasso}
    \begin{align*}
      \hat\beta =& \arg\min_\beta \left\{ \sum_{i = 1}^n \left(y_i - \beta_0 - \sum_{j = 1}^p \beta_j x_{ij}\right)^2 \right\}\\
      & {\color{blue} \text{subject to } \sum_{j = 1}^p |\beta_j| \le s}
    \end{align*}
    \vfill
    Or, equivalently:
    \begin{align*}
      \hat\beta =& \arg\min_\beta \left\{ \sum_{i = 1}^n \left(y_i - \beta_0 - \sum_{j = 1}^p \beta_j x_{ij}\right)^2 + {\color{blue}\lambda \sum_{j = 1}^p |\beta_j|} \right\}
    \end{align*}
  \end{frame}

  \begin{frame}{Ridge Regression}
    Lasso:
    \begin{align*}
      \hat\beta =& \arg\min_\beta \left\{ \sum_{i = 1}^n \left(y_i - \beta_0 - \sum_{j = 1}^p \beta_j x_{ij}\right)^2 + \lambda \sum_{j = 1}^p {\color{blue} |\beta_j|} \right\}
    \end{align*}
    \vfill
    Ridge Regression:
    \begin{align*}
      \hat\beta =& \arg\min_\beta \left\{ \sum_{i = 1}^n \left(y_i - \beta_0 - \sum_{j = 1}^p \beta_j x_{ij}\right)^2 + \lambda \sum_{j = 1}^p {\color{blue}\beta_j^2} \right\}
    \end{align*}
  \end{frame}

  \begin{frame}[fragile=singleslide]\frametitle{Elastic Net Regression}
    \begin{align*}
      \sum_{i = 1}^n w_i \left(y_i - \beta_0 - \sum_{j = 1}^p \beta_j x_{ij}\right)^2 + (1 - \alpha) \cdot \lambda \sum_{j = 1}^p \beta_j^2 + \alpha \cdot \lambda \sum_{j = 1}^p |\beta_j|
    \end{align*}
    \begin{columns}
      \begin{column}{0.5\linewidth}
        \begin{verbatim}
    glmnet::glmnet(
      x,
      y,
      weights = NULL,
      alpha = 1,
      lambda = NULL,
      standardize = TRUE,
      intercept = TRUE,
      ...
    )
      \end{verbatim}
      \end{column}
      \begin{column}{0.5\linewidth}

      nfolds = 10,
      foldid = NULL,
      parallel = FALSE,
      ...
    )
      \end{verbatim}
